title: Usage
description: Basic usage guide and interactive press handler examples for TajweedVerse.
---

Pass raw verse text with markup tags to `<TajweedVerse />`:

```tsx
import React from 'react';
import { Alert } from 'react-native';
import TajweedVerse from '@masumdev/rn-tajweed-verse';

const sampleVerse = "[h[bi[s[m[i [l[al[l[h[i [n[al[r[n[h[n[m[n[n[i [n[al[r[n[h[i[m[i";

export function VerseExample() {
  return (
    <TajweedVerse
      text={sampleVerse}
      style={{ fontSize: 24, textAlign: 'right', fontFamily: 'Amiri' }}
      config={{
        qlq: {
          style: { color: '#2b6cb0', fontWeight: 'bold' },
          onPress: (matchedText) => Alert.alert('Qalqalah Rule', matchedText),
        },
        madda_normal: {
          style: { color: '#c53030' },
        },
      }}
    />
  );
}
```
