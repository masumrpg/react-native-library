Render a QR code component with custom styling:

```tsx
import React from 'react';
import QRCodeGen from '@masumdev/react-native-qr-code-gen';

export function QRCodeExample() {
  return (
    <QRCodeGen
      value="https://masumdev.vercel.app"
      size={240}
      color="#1e293b"
      backgroundColor="#ffffff"
      eyeOptions={{
        shape: 'circle',
        color: '#2563eb',
      }}
      pieceOptions={{
        shape: 'rounded',
        color: '#1e293b',
      }}
    />
  );
}
```
