## Using the `useTheme` Hook

Access theme colors, mode state, and toggle functions from any child component:

```tsx
import React from 'react';
import { useTheme, Button, Text, Box } from '@masumdev/rn-ui';

export function ThemeToggleExample() {
  const { colors, isDark, colorScheme, setColorScheme, toggleColorScheme } = useTheme();

  return (
    <Box style={{ backgroundColor: colors.background, padding: 16 }}>
      <Text style={{ color: colors.text }}>
        Current Mode: {colorScheme} (Active: {isDark ? 'Dark' : 'Light'})
      </Text>
      <Button tone="secondary" onPress={toggleColorScheme}>
        Toggle Theme
      </Button>
    </Box>
  );
}
```

## Persistent Storage Integration

Pass `@react-native-async-storage/async-storage` or `expo-secure-store` to automatically persist user theme choices across app reboots:

```tsx title="App.tsx"
import AsyncStorage from '@react-native-async-storage/async-storage';
import { ThemeProvider } from '@masumdev/rn-ui';

export default function App() {
  return (
    <ThemeProvider
      defaultColorScheme="system"
      storage={AsyncStorage}
      storageKey="user-theme-pref"
    >
      <RootNavigation />
    </ThemeProvider>
  );
}
```
