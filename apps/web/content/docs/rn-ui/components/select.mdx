import { Select } from '@masumdev/rn-ui';

<Select
  placeholder="Choose a language"
  options={[
    { label: 'English', value: 'en' },
    { label: 'Bahasa Indonesia', value: 'id' },
    { label: 'Arabic', value: 'ar' },
  ]}
  onValueChange={(val) => console.log(val)}
/>
