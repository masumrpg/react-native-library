---
title: FormField
description: Form field wrapper combining Label, Input, and helper/error text messages.
---

```tsx
import { FormField, Input } from '@masumdev/rn-ui';

<FormField label="Email Address" error="Invalid email address">
  <Input placeholder="user@example.com" />
</FormField>
```
