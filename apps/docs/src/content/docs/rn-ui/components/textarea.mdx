export const exampleCode = `import { useState } from 'react';
import { Textarea } from '@masumdev/rn-ui';

export function TextareaExample() {
  const [value, setValue] = useState('');

  return (
    <Textarea
      placeholder="Write a message or feedback..."
      value={value}
      onChangeText={setValue}
      numberOfLines={4}
    />
  );
}`;

`Textarea` renders multiline text input boxes for descriptions and feedback forms.
