export function PopoverExample() {
  return (
    <Popover>
      <PopoverTrigger>
        <Button variant="outline">Open Popover</Button>
      </PopoverTrigger>
      <PopoverContent>
        <Text>Floating card overlay content</Text>
      </PopoverContent>
    </Popover>
  );
}`;

`Popover` renders floating content popups positioned relative to anchor triggers.
