export const exampleCode = `import { useState } from 'react';
import { RadioGroup, RadioGroupItem } from '@masumdev/rn-ui';

export function RadioExample() {
  const [value, setValue] = useState('expo');

  return (
    <RadioGroup value={value} onValueChange={setValue}>
      <RadioGroupItem value="expo" label="Expo Framework" />
      <RadioGroupItem value="cli" label="React Native CLI" />
    </RadioGroup>
  );
}`;

`RadioGroup` allows users to select a single option from a set.
