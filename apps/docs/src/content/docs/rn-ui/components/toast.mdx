export const toastUsageCode = `import React from 'react';
import { useToast, Button, Box } from '@masumdev/rn-ui';

export function ToastExample() {
  const toast = useToast();

  const handleShowToast = () => {
    toast.show({
      title: 'Action Completed',
      description: 'Your settings have been saved successfully.',
      tone: 'success',
      duration: 3500,
      action: {
        label: 'Undo',
        onPress: () => console.log('Undo pressed'),
      },
    });
  };

  return (
    <Box gap="sm">
      <Button tone="success" onPress={handleShowToast}>
        Trigger Success Toast
      </Button>
      <Button tone="danger" variant="outline" onPress={() => toast.dismiss()}>
        Dismiss All Toasts
      </Button>
    </Box>
  );
}`;

export const providerCode = `import { ToastProvider } from '@masumdev/rn-ui';

export default function App() {
  return (
    <ToastProvider placement="bottom" maxToasts={3}>
      <RootNavigation />
    </ToastProvider>
  );
}`;

`Toast` provides programmatic, animated notification toasts with swipe-to-dismiss support, action buttons, and automatic timers.

## Usage with `useToast()`

Call `useToast()` inside any child component of `ToastProvider`:

<Code code={toastUsageCode} lang="tsx" title="ToastExample.tsx" mark={['useToast', 'toast.show', 'toast.dismiss']} />

Wrap your application root with `ToastProvider`:

## `useToast()` API

Calling `useToast()` returns an object with the following methods:

<TypeTable
  type={{
    show: {
      type: '(options: ToastOptions) => string',
      description: 'Triggers a new toast notification and returns its unique ID.',
    },
    dismiss: {
      type: '(id?: string) => void',
      description: 'Dismisses a specific toast by ID, or all toasts if no ID is passed.',
    },
    update: {
      type: '(id: string, options: ToastOptions) => void',
      description: 'Updates properties of an existing active toast in place.',
    },
  }}
/>
