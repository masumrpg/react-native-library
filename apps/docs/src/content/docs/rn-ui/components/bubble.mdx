export function ChatExample() {
  return (
    <BubbleGroup>
      <Bubble align="left">
        <BubbleContent>Hello! How can I help you today?</BubbleContent>
      </Bubble>
      <Bubble align="right" variant="filled">
        <BubbleContent>I'd like to check the RN UI docs.</BubbleContent>
      </Bubble>
    </BubbleGroup>
  );
}`;

`Bubble` renders chat message bubbles styled for messaging interfaces.
