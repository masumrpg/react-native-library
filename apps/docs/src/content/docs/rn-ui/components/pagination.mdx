export const exampleCode = `import { useState } from 'react';
import { Pagination } from '@masumdev/rn-ui';

export function PaginationExample() {
  const [page, setPage] = useState(1);

  return (
    <Pagination
      currentPage={page}
      totalPages={10}
      onPageChange={setPage}
    />
  );
}`;

`Pagination` enables navigation through multi-page data lists.
