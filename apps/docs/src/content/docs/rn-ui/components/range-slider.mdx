export const exampleCode = `import { useState } from 'react';
import { RangeSlider } from '@masumdev/rn-ui';

export function RangeSliderExample() {
  const [range, setRange] = useState<[number, number]>([200, 800]);

  return (
    <RangeSlider
      min={0}
      max={1000}
      step={50}
      value={range}
      onValueChange={setRange}
      tone="primary"
    />
  );
}`;

`RangeSlider` provides a dual-thumb slider for selecting a value range. Built with React Native Gesture Handler and Reanimated for direct 1:1 gesture tracking.
