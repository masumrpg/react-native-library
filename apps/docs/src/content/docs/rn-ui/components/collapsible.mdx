export const exampleCode = `import { useState } from 'react';
import { Collapsible, CollapsibleTrigger, CollapsibleContent, Button, Text } from '@masumdev/rn-ui';

export function CollapsibleExample() {
  const [open, setOpen] = useState(false);

  return (
    <Collapsible open={open} onOpenChange={setOpen}>
      <CollapsibleTrigger>
        <Button variant="outline">{open ? 'Hide Details' : 'Show Details'}</Button>
      </CollapsibleTrigger>
      <CollapsibleContent>
        <Text>Hidden collapsible content disclosed on press.</Text>
      </CollapsibleContent>
    </Collapsible>
  );
}`;

`Collapsible` toggles the visibility of an element.
