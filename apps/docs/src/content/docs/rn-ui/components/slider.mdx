export const exampleCode = `import { useState } from 'react';
import { Slider } from '@masumdev/rn-ui';

export function SliderExample() {
  const [val, setVal] = useState(50);

  return (
    <Slider
      value={val}
      onValueChange={setVal}
      minimumValue={0}
      maximumValue={100}
    />
  );
}`;

`Slider` allows users to select a value from a continuous or discrete range.
