export const exampleCode = `import { useState } from 'react';
import { Stepper } from '@masumdev/rn-ui';

export function StepperExample() {
  const [step, setStep] = useState(2);

  return (
    <Stepper
      currentStep={step}
      totalSteps={4}
      onStepChange={setStep}
    />
  );
}`;

`Stepper` tracks progress across multi-step wizard screens.
