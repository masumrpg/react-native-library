import { Code } from '@astrojs/starlight/components';
import TypeTable from '@/components/TypeTable.astro';

export const formFieldCode = `import React from 'react';
import { FormField, FormLabel, FormControl, FormDescription, FormMessage, Input } from '@masumdev/rn-ui';

export function FormFieldExample() {
  return (
    <FormField invalid required>
      <FormLabel>Username</FormLabel>
      <FormControl>
        <Input placeholder="enter_username" error />
      </FormControl>
      <FormDescription>Choose a unique public handle.</FormDescription>
      <FormMessage>Username is already taken.</FormMessage>
    </FormField>
  );
}`;

`FormField` manages form context (invalid state, required asterisk, disabled) for labels, input controls, helper descriptions, and validation error messages.

## Usage

## Props API

### FormField

<TypeTable
  type={{
    invalid: { type: 'boolean', default: 'false', description: 'Highlights error state across child message and label.' },
    required: { type: 'boolean', default: 'false', description: 'Applies required asterisk (*) to FormLabel.' },
    disabled: { type: 'boolean', default: 'false', description: 'Disables user interaction.' },
  }}
/>

### FormLabel

Inherits all props from `<Label />`.
