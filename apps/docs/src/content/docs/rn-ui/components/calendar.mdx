export const exampleCode = `import React, { useState } from 'react';
import { Calendar, Box, Text, Badge } from '@masumdev/rn-ui';

export function CalendarExample() {
  const [selected, setSelected] = useState('2026-07-12');

  // Sample event markings with colored dots
  const markedEvents = {
    [selected]: { selected: true },
    '2026-07-05': { marked: true, dotColor: '#06B6D4' },
    '2026-07-12': {
      dots: [
        { key: 'meeting', color: '#06B6D4' },
        { key: 'task', color: '#F59E0B' },
      ],
    },
    '2026-07-15': { marked: true, dotColor: '#10B981' },
  };

  return (
    <Box gap="md">
      <Calendar
        current={selected}
        markedDates={markedEvents}
        onDayPress={(day) => setSelected(day.dateString)}
        showTodayButton
        enableYearMonthPicker
      />
      <Text color="textMuted" variant="bodySmall">
        Selected Date: <Text color="primary" weight="600">{selected}</Text>
      </Text>
    </Box>
  );
}`;

`Calendar` provides interactive month view date selection with event dot indicators, quick "Today" navigation, fast month/year overlay selectors, and period range selection.

  type={{

    onDayPress: { type: '(day: { dateString: string }) => void', description: 'Callback triggered when a calendar day cell is pressed.' },

  }}
