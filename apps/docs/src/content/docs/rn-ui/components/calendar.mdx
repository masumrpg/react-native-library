export const exampleCode = `import { useState } from 'react';
import { Calendar } from '@masumdev/rn-ui';

export function CalendarExample() {
  const [selected, setSelected] = useState('2026-08-23');

  return (
    <Calendar
      current={selected}
      onDayPress={(day) => setSelected(day.dateString)}
      markedDates={{
        [selected]: { selected: true, selectedColor: '#06B6D4' },
      }}
    />
  );
}`;

`Calendar` provides interactive month view date selection and range marking powered by `react-native-calendars`.
