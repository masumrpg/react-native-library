import { Code } from '@astrojs/starlight/components';
import TypeTable from '@/components/TypeTable.astro';

export const exampleCode = `import { TimelineCalendar, CalendarProvider } from '@masumdev/rn-ui';

const events = [
  {
    start: '2026-08-25 09:00',
    end: '2026-08-25 10:30',
    title: 'Team Standup',
    summary: 'Daily sync meeting',
    color: '#6366F1',
  },
];

export function TimelineExample() {
  return (
    <CalendarProvider date="2026-08-25">
      <TimelineCalendar events={events} />
    </CalendarProvider>
  );
}`;

`TimelineCalendar` renders a day/week schedule timeline with hour lanes and packed event blocks. Ideal for scheduling and calendar app interfaces.

## Usage

<Code code={exampleCode} lang="tsx" title="TimelineExample.tsx" mark={['TimelineCalendar', 'events']} />
