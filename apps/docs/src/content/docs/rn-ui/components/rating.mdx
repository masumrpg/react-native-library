import { Code } from '@astrojs/starlight/components';

export const exampleCode = `import { useState } from 'react';
import { Rating } from '@masumdev/rn-ui';

export function RatingExample() {
  const [rating, setRating] = useState(4.5);

  return (
    <Rating
      value={rating}
      onRatingChange={setRating}
      maxCount={5}
      precision={0.5}
    />
  );
}`;

`Rating` displays star or custom shape ratings with interactive tap and drag input.
