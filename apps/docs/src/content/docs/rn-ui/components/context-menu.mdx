export const exampleCode = `import { ContextMenu, ContextMenuTrigger, ContextMenuContent, ContextMenuItem, Text } from '@masumdev/rn-ui';

export function ContextMenuExample() {
  return (
    <ContextMenu>
      <ContextMenuTrigger>
        <Text>Long press this text to open context menu</Text>
      </ContextMenuTrigger>
      <ContextMenuContent>
        <ContextMenuItem onPress={() => console.log('Copy')}>Copy Text</ContextMenuItem>
        <ContextMenuItem onPress={() => console.log('Share')}>Share</ContextMenuItem>
      </ContextMenuContent>
    </ContextMenu>
  );
}`;
