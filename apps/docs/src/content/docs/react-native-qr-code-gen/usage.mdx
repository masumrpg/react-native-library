export const basicUsageCode = `import React from 'react';
import { QRCode } from '@masumdev/react-native-qr-code-gen';

export function BasicQRCodeExample() {
  return (
    <QRCode
      value="https://masum.dev"
      size={200}
      variant="BASIC"
    />
  );
}`;

export const presetVariantCode = `import React from 'react';
import { QRCode } from '@masumdev/react-native-qr-code-gen';

export function PresetVariantsExample() {
  return (
    <>
      {/* Heart preset variant */}
      <QRCode value="Love QR" size={180} variant="HEART" />

      {/* Linear Gradient preset variant */}
      <QRCode value="Gradients" size={180} variant="LINEAR_GRADIENT" />

      {/* Triangle preset variant */}
      <QRCode value="Triangle" size={140} variant="TRIANGLE" />
    </>
  );
}`;

export const loadingStateCode = `import React, { useState } from 'react';
import { View, Text } from 'react-native';
import { QRCode } from '@masumdev/react-native-qr-code-gen';

export function LoadingQRCodeExample() {
  const [loading, setLoading] = useState(true);

  return (
    <QRCode
      value="https://masum.dev"
      size={180}
      isLoading={loading}
      renderLoading={() => (
        <View style={{ width: 180, height: 180, justifyContent: 'center', alignItems: 'center' }}>
          <Text>Generating QR Code...</Text>
        </View>
      )}
    />
  );
}`;
