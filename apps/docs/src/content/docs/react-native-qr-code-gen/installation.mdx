export const npmInstall = `npm i @masumdev/react-native-qr-code-gen react-native-svg`;
export const pnpmInstall = `pnpm add @masumdev/react-native-qr-code-gen react-native-svg`;
export const yarnInstall = `yarn add @masumdev/react-native-qr-code-gen react-native-svg`;
export const bunInstall = `bun add @masumdev/react-native-qr-code-gen react-native-svg`;
